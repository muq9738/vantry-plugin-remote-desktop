cmake_minimum_required(VERSION 3.24)
project(vantry_windows_media_helper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(WIN32 AND MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(BUILD_TESTING)
  add_executable(protocol-core-tests tests/core_tests.cpp)
  target_include_directories(protocol-core-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME protocol-core COMMAND protocol-core-tests)
endif()
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Use MSVC or clang-cl with the Windows SDK")
  endif()
  add_executable(vantry-remote-helper main.cpp)
  target_compile_definitions(vantry-remote-helper PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
  target_compile_options(vantry-remote-helper PRIVATE /W4 /permissive- /EHsc /utf-8 /guard:cf /sdl)
  target_link_options(vantry-remote-helper PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /MANIFEST:EMBED /MANIFESTUAC:NO /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/helper.manifest)
  target_link_libraries(vantry-remote-helper PRIVATE d3d11 dxgi mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32 uuid strmiids wtsapi32 user32)
  install(TARGETS vantry-remote-helper RUNTIME DESTINATION .)
else()
  message(STATUS "Non-Windows host: only portable protocol tests are built; no media runtime is available")
endif()
